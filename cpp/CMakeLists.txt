cmake_minimum_required(VERSION 3.15)
project(silero_vad_axera_cpp LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include_directories(include)
add_library(silero_vad_axera_sdk STATIC src/model_runner.cpp src/silero_vad.cpp)
target_include_directories(silero_vad_axera_sdk PUBLIC include)
add_executable(silero_vad_example examples/main.cpp)
if(AX_RUNTIME_ROOT)
  target_include_directories(silero_vad_axera_sdk PUBLIC ${AX_RUNTIME_ROOT}/include)
  target_link_directories(silero_vad_axera_sdk PUBLIC ${AX_RUNTIME_ROOT}/lib)
  target_link_libraries(silero_vad_axera_sdk PUBLIC ax_engine ax_sys ax_interpreter)
endif()
target_link_libraries(silero_vad_example PRIVATE silero_vad_axera_sdk)
